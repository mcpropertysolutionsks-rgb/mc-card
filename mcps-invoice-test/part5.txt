<div class="meta">${esc(i.property||"No property")} · ${esc(i.invoiceDate||"")}</div>
      <div class="meta">${money(totalFor(i))}</div>
      <span class="status">${esc(i.status||"Draft")}</span>
    </div>`).join("") || `<div class="empty">No invoices found.</div>`;
  document.querySelectorAll(".inv-card").forEach(el=>el.onclick=async()=>{
    saveFormIntoCurrent();
    state.currentId=el.dataset.id; persist();
    const inv=current();
    try{
      if(!(inv.photos||[]).some(p=>p.data)){
        const dbPhotos=await loadPhotosDB(inv.id);
        if(dbPhotos.length) inv.photos=dbPhotos;
      }
    }catch(e){console.warn(e)}
    fillForm(inv); renderAll();
  });
}

function renderLineEditor(inv){
  $("lineEditorBody").innerHTML=(inv.lines||[]).map((l,idx)=>`
    <tr data-line="${idx}">
      <td><input data-k="description" value="${esc(l.description)}" placeholder="Service name"></td>
      <td><input data-k="details" value="${esc(l.details)}" placeholder="Short description"></td>
      <td><input data-k="qty" value="${Number(l.qty||1)}" type="number" step="0.01" class="w-small"></td>
      <td><input data-k="cost" value="${Number(l.cost||0)}" type="number" step="0.01" class="w-small"></td>
      <td><button class="remove" title="Remove">×</button></td>
    </tr>`).join("");
  document.querySelectorAll("#lineEditorBody tr").forEach(tr=>{
    tr.querySelectorAll("input").forEach(inp=>inp.addEventListener("input",()=>{
      const idx=Number(tr.dataset.line); const key=inp.dataset.k;
      inv.lines[idx][key]=(key==="cost"||key==="qty") ? Number(inp.value||0) : inp.value;
      updateMoneySummary(); renderPreview();
    }));
    tr.querySelector(".remove").onclick=()=>{
      const idx=Number(tr.dataset.line); inv.lines.splice(idx,1);
      if(!inv.lines.length) inv.lines.push({description:"",details:"",cost:0,qty:1});
      renderLineEditor(inv); updateMoneySummary(); renderPreview();
    };
  });
}
function renderPhotoEditor(inv){
  workingPhotos=inv.photos||[];
  if($("photoCount")) $("photoCount").textContent=`Photos attached: ${workingPhotos.length}`;
  $("photoEditorGrid").innerHTML=workingPhotos.map((p,idx)=>`
    <div class="photo-edit">
      <img src="${p.data}">
      <input value="${esc(p.caption||"")}" data-photo="${idx}" placeholder="Photo caption">
      <button class="remove-photo" data-photo="${idx}" style="margin-top:6px;border:0;background:#eee;border-radius:6px;padding:6px 8px">Remove</button>
    </div>`).join("");
  document.querySelectorAll("[data-photo]").forEach(el=>{
    if(el.tagName==="INPUT") el.oninput=()=>{inv.photos[Number(el.dataset.photo)].caption=el.value;renderPreview()};
  });
  document.querySelectorAll(".remove-photo").forEach(b=>b.onclick=async()=>{
    const idx=Number(b.dataset.photo); const p=inv.photos[idx];
    if(p?.id){try{await deletePhotoDB(inv.id,p.id)}catch(e){console.warn(e)}}
    inv.photos.splice(idx,1);renderPhotoEditor(inv);renderPreview();persist();
  });
}
function fillForm(inv){
  ["manager","property","address","clientPhone","clientEmail","invoiceNo","status","workOrder","unit","building","invoiceDate","dueDate","taxRate","companyPhone","companyEmail","companyWebsite"].forEach(k=>{
    if($(k)) $(k).value=inv[k] ?? "";
  });
  renderLineEditor(inv); renderPhotoEditor(inv); updateMoneySummary(); renderPreview();
}
function saveFormIntoCurrent(){
  const inv=current(); if(!inv) return;
  ["manager","property","address","clientPhone","clientEmail","invoiceNo","status","workOrder","unit","building","invoiceDate","dueDate","companyPhone","companyEmail","companyWebsite"].forEach(k=>inv[k]=$(k).value);
  inv.taxRate=Number($("taxRate").value||0);
  persist();
}
function subtotalFor(inv){return (inv.lines||[]).reduce((s,l)=>s+Number(l.cost||0)*Number(l.qty||0),0)}
function taxFor(inv){return subtotalFor(inv)*Number(inv.taxRate||0)/100}
function totalFor(inv){return subtotalFor(inv)+taxFor(inv)}
function updateMoneySummary(){
  const inv=current(); if(!inv)return;
  $("moneySummary").textContent=`Subtotal ${money(subtotalFor(inv))} · Tax ${money(taxFor(inv))} · Total ${money(totalFor(inv))}`;
}
