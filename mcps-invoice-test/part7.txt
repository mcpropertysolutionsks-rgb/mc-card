async function fileToDataURL(file){
  return await new Promise((resolve,reject)=>{
    const r=new FileReader();
    r.onload=()=>resolve(r.result);
    r.onerror=reject;
    r.readAsDataURL(file);
  });
}
async function normalizePhotoFile(file){
  const name=(file.name||"").toLowerCase();
  const type=(file.type||"").toLowerCase();
  const isHeic=type.includes("heic")||type.includes("heif")||name.endsWith(".heic")||name.endsWith(".heif");

  if(isHeic){
    if(typeof heic2any!=="function"){
      throw new Error("HEIC_CONVERTER_UNAVAILABLE");
    }
    const converted=await heic2any({blob:file,toType:"image/jpeg",quality:.82});
    const blob=Array.isArray(converted)?converted[0]:converted;
    return new File([blob],(file.name||"photo").replace(/\.(heic|heif)$/i,".jpg"),{type:"image/jpeg"});
  }
  return file;
}
async function resizePhoto(file){
  const normalized=await normalizePhotoFile(file);
  const data=await fileToDataURL(normalized);
  const img=await new Promise((resolve,reject)=>{
    const im=new Image();
    im.onload=()=>resolve(im);
    im.onerror=()=>reject(new Error("IMAGE_DECODE_FAILED"));
    im.src=data;
  });
  const maxW=1000,maxH=750,scale=Math.min(1,maxW/img.width,maxH/img.height);
  const c=document.createElement("canvas");
  c.width=Math.max(1,Math.round(img.width*scale));
  c.height=Math.max(1,Math.round(img.height*scale));
  c.getContext("2d").drawImage(img,0,0,c.width,c.height);
  return c.toDataURL("image/jpeg",.70);
}
$("photoInput").onchange=async e=>{
  const inv=current(); const files=Array.from(e.target.files||[]);
  if(!files.length)return;
  let added=0,failed=[];
  for(const f of files){
    try{
      const data=await resizePhoto(f);
      const photo={id:uid(),data,caption:"Completed work"};
      inv.photos.push(photo);
      try{await putPhoto(inv.id,photo)}catch(err){console.warn("Photo DB save failed",err)}
      added++;
    }catch(err){
      console.error("Photo conversion failed",f.name,err);
      failed.push(f.name||"photo");
    }
  }
  e.target.value="";
  persist();
  renderPhotoEditor(inv);
  renderPreview();

  if(added && !failed.length){
    alert(`${added} photo${added===1?"":"s"} added. Preview now includes the photo page${added===1?"":"s"}.`);
  }else if(added && failed.length){
    alert(`${added} photo${added===1?"":"s"} added. ${failed.length} could not be converted: ${failed.join(", ")}`);
  }else{
    alert("The selected photo could not be converted. If it is HEIC/HEIF, make sure the device is online so the converter can load, or change the camera setting to JPG for this test.");
  }
};
