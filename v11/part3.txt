  return new Promise((resolve,reject)=>{
    const tx=db.transaction(PHOTO_STORE,"readwrite");
    tx.objectStore(PHOTO_STORE).put({key:`${invoiceId}:${photo.id}`,invoiceId,id:photo.id,data:photo.data,caption:photo.caption||""});
    tx.oncomplete=()=>{db.close();resolve()};
    tx.onerror=()=>{db.close();reject(tx.error)};
  });
}
async function deletePhotoDB(invoiceId, photoId){
  const db=await openPhotoDB();
  return new Promise((resolve,reject)=>{
    const tx=db.transaction(PHOTO_STORE,"readwrite");
    tx.objectStore(PHOTO_STORE).delete(`${invoiceId}:${photoId}`);
    tx.oncomplete=()=>{db.close();resolve()};
    tx.onerror=()=>{db.close();reject(tx.error)};
  });
}
async function loadPhotosDB(invoiceId){
  const db=await openPhotoDB();
  return new Promise((resolve,reject)=>{
    const tx=db.transaction(PHOTO_STORE,"readonly");
    const req=tx.objectStore(PHOTO_STORE).getAll();
    req.onsuccess=()=>{db.close();resolve(req.result.filter(x=>x.invoiceId===invoiceId).map(x=>({id:x.id,data:x.data,caption:x.caption||""})))};
    req.onerror=()=>{db.close();reject(req.error)};
  });
}
async function deleteInvoicePhotosDB(invoiceId){
  const all=await loadPhotosDB(invoiceId);
  for(const p of all) await deletePhotoDB(invoiceId,p.id);
}

function todayISO(){
  const d=new Date();
  const z=n=>String(n).padStart(2,"0");
  return `${d.getFullYear()}-${z(d.getMonth()+1)}-${z(d.getDate())}`;
}
function plusDays(iso,days){
  const d=new Date(iso+"T12:00:00"); d.setDate(d.getDate()+days);
  return d.toISOString().slice(0,10);
}
function nextInvoiceNumber(){
  let c = Number(localStorage.getItem(COUNTER_KEY)||0)+1;
  localStorage.setItem(COUNTER_KEY,String(c));
  return `MCPS-${new Date().getFullYear()}-${String(c).padStart(4,"0")}`;
}
function uid(){return (crypto.randomUUID ? crypto.randomUUID() : Date.now()+"-"+Math.random().toString(16).slice(2));}

function loadState(){
  try{
    const saved=JSON.parse(localStorage.getItem(STORAGE_KEY)||"null");
    if(saved && Array.isArray(saved.invoices)){state=saved}
  }catch(e){}
  if(!state.invoices.length){
    const inv=defaultInvoice();
    state.invoices=[inv]; state.currentId=inv.id; persist();
  } else if(!state.currentId || !state.invoices.some(i=>i.id===state.currentId)){
    state.currentId=state.invoices[0].id;
  }
}
function persist(){
  try{
    localStorage.setItem(STORAGE_KEY,JSON.stringify(state));
  }catch(e){
    console.warn("Local storage is full; invoice remains available in this session.",e);
  }
}

function current(){return state.invoices.find(i=>i.id===state.currentId)}

function renderList(){
  const q=$("searchInvoices").value.toLowerCase().trim();
  const list=state.invoices
    .slice().sort((a,b)=>(b.invoiceDate||"").localeCompare(a.invoiceDate||""))
    .filter(i=>[i.invoiceNo,i.property,i.manager,i.workOrder,i.status].join(" ").toLowerCase().includes(q));
  $("invoiceList").innerHTML=list.map(i=>`
    <div class="inv-card ${i.id===state.currentId?"active":""}" data-id="${i.id}">
      <div class="num">${esc(i.invoiceNo||"Untitled")}</div>
      <div class="meta">${esc(i.property||"No property")} · ${esc(i.invoiceDate||"")}</div>
      <div class="meta">${money(totalFor(i))}</div>
      <span class="status">${esc(i.status||"Draft")}</span>
    </div>`).join("") || `<div class="empty">No invoices found.</div>`;
  document.querySelectorAll(".inv-card").forEach(el=>el.onclick=async()=>{
    saveFormIntoCurrent();
    state.currentId=el.dataset.id; persist();
    const inv=current();
    try{
      if(!(inv.photos||[]).some(p=>p.data)){
        const dbPhotos=await loadPhotosDB(inv.id);
        if(dbPhotos.length) inv.photos=dbPhotos;
      }
    }catch(e){console.warn(e)}
    fillForm(inv); renderAll();
  });
}

    tr.querySelectorAll("input").forEach(inp=>inp.addEventListener("input",()=>{
      const idx=Number(tr.dataset.line); const key=inp.dataset.k;
      inv.lines[idx][key]=(key==="cost"||key==="qty") ? Number(inp.value||0) : inp.value;

function saveFormIntoCurrent(){
  const inv=current(); if(!inv) return;
  ["manager","property","address","clientPhone","clientEmail","invoiceNo","status","workOrder","unit","building","invoiceDate","dueDate","companyPhone","companyEmail","companyWebsite"].forEach(k=>inv[k]=$(k).value);
  inv.taxRate=Number($("taxRate").value||0);
  persist();
}

function taxFor(inv){return subtotalFor(inv)*Number(inv.taxRate||0)/100}
function totalFor(inv){return subtotalFor(inv)+taxFor(inv)}
